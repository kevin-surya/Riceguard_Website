## 3. Modelling dan hyperparameter tuning

Setiap kandidat di-tuning menggunakan enam observasi terakhir dari data train sebagai validasi internal (2013–2018). Sesudah itu, konfigurasi terpilih di-fit ulang pada seluruh data train dan diuji sekali pada enam observasi test.


--- CELL 8 code ---
def mape(y_true, y_pred):
    y_true, y_pred = np.asarray(y_true), np.asarray(y_pred)
    return np.mean(np.abs((y_true - y_pred) / y_true)) * 100

def metrics(y_true, y_pred):
    return {"MAPE (%)": mape(y_true, y_pred),
            "RMSE": np.sqrt(mean_squared_error(y_true, y_pred)),
            "MAE": mean_absolute_error(y_true, y_pred)}

def supervised(values, lag):
    values = np.asarray(values, dtype=float)
    return np.array([values[i-lag:i] for i in range(lag, len(values))]), values[lag:]

def fc_des(y, h, p):
    return Holt(y, exponential=p["exponential"], damped_trend=p["damped_trend"]).fit(optimized=True).forecast(h)

def fc_arima(y, h, p):
    return ARIMA(y, order=p["order"], trend=p["trend"]).fit().forecast(h)

def fc_mlp(y, h, p):
    scaler = MinMaxScaler()
    z = scaler.fit_transform(np.asarray(y).reshape(-1, 1)).ravel()
    X, target = supervised(z, p["lag"])
    model = MLPRegressor(hidden_layer_sizes=p["hidden"], alpha=p["alpha"],
                         learning_rate_init=p["lr"], max_iter=3000,
                         early_stopping=True, validation_fraction=.2,
                         random_state=SEED)
    model.fit(X, target)
    hist, pred = list(z), []
    for _ in range(h):
        v = model.predict(np.array(hist[-p["lag"]:]).reshape(1, -1))[0]
        hist.append(v); pred.append(v)
    return scaler.inverse_transform(np.array(pred).reshape(-1, 1)).ravel()

class TinyLSTM(nn.Module):
    def __init__(self, hidden):
        super().__init__()
        self.lstm = nn.LSTM(1, hidden, batch_first=True)
        self.out = nn.Linear(hidden, 1)
    def forward(self, x):
        return self.out(self.lstm(x)[0][:, -1, :])

def fc_lstm(y, h, p):
    scaler = MinMaxScaler()
    z = scaler.fit_transform(np.asarray(y).reshape(-1, 1)).ravel()
    X, target = supervised(z, p["lag"])
    x = torch.tensor(X[:, :, None], dtype=torch.float32)
    target = torch.tensor(target[:, None], dtype=torch.float32)
    torch.manual_seed(SEED)
    model = TinyLSTM(p["hidden"])
    opt = torch.optim.Adam(model.parameters(), lr=p["lr"], weight_decay=p["weight_decay"])
    loss_fn = nn.MSELoss()
    model.train()
    for _ in range(p["epochs"]):
        opt.zero_grad(); loss = loss_fn(model(x), target); loss.backward(); opt.step()
    model.eval()
    hist, pred = list(z), []
    with torch.no_grad():
        for _ in range(h):
            a = torch.tensor(np.array(hist[-p["lag"]:])[None, :, None], dtype=torch.float32)
            v = model(a).item(); hist.append(v); pred.append(v)
    return scaler.inverse_transform(np.array(pred).reshape(-1, 1)).ravel()

def fc_xgb(y, h, p):
    X, target = supervised(y, p["lag"])
    model = XGBRegressor(n_estimators=p["n_estimators"], max_depth=p["max_depth"],
                         learning_rate=p["learning_rate"], subsample=.9,
                         colsample_bytree=.9, objective="reg:squarederror",
                         random_state=SEED, n_jobs=1)
    model.fit(X, target)
    hist, pred = list(np.asarray(y, dtype=float)), []
    for _ in range(h):
        v = model.predict(np.array(hist[-p["lag"]:]).reshape(1, -1))[0]
        hist.append(v); pred.append(v)
    return np.array(pred)

def fc_prophet(y, years, h, p):
    frame = pd.DataFrame({"ds": pd.to_datetime(np.asarray(years).astype(str) + "-01-01"), "y": y})
    model = Prophet(growth="linear", yearly_seasonality=False, weekly_seasonality=False,
                    daily_seasonality=False, changepoint_prior_scale=p["cps"],
                    seasonality_mode=p["mode"])
    model.fit(frame)
    future = model.make_future_dataframe(periods=h, freq="YS")
    return model.predict(future)["yhat"].tail(h).to_numpy()

FORECASTERS = {"Double Exponential Smoothing": fc_des, "ARIMA": fc_arima,
               "Multilayer Perceptron": fc_mlp, "LSTM": fc_lstm,
               "XGBoost": fc_xgb, "Prophet": fc_prophet}

GRIDS = {
 "Double Exponential Smoothing": [{"exponential": e, "damped_trend": d} for e in [False, True] for d in [False, True]],
 "ARIMA": [{"order": (p,d,q), "trend": t} for p in range(3) for d in [0,1] for q in range(3) for t in ["n","t"] if not (d > 0 and t == "t")],
 "Multilayer Perceptron": [{"lag": l, "hidden": h, "alpha": a, "lr": lr} for l in [3,5] for h in [(8,), (16,)] for a in [.0001,.01] for lr in [.001,.01]],
 "LSTM": [{"lag": l, "hidden": h, "lr": lr, "weight_decay": wd, "epochs": 250} for l in [3,5] for h in [8,16] for lr in [.005,.01] for wd in [0,.0001]],
 "XGBoost": [{"lag": l, "n_estimators": n, "max_depth": d, "learning_rate": lr} for l in [3,5] for n in [50,100] for d in [2,3] for lr in [.05,.1]],
 "Prophet": [{"cps": cps, "mode": mode} for cps in [.01,.1,.5] for mode in ["additive","multiplicative"]]
}

# Uji Diebold-Mariano: LSTM dibandingkan dengan setiap model lain.
# d = loss_LSTM - loss_pembanding; DM < 0 berarti loss LSTM lebih kecil.
def diebold_mariano(error_1, error_2, horizon=1):
    error_1, error_2 = np.asarray(error_1, dtype=float), np.asarray(error_2, dtype=float)
    n = len(error_1)
    d = error_1**2 - error_2**2  # squared-error loss differential
    max_lag = min(horizon - 1, n - 1)
    long_run_var = np.var(d, ddof=1)
    for lag in range(1, max_lag + 1):
        covariance = np.cov(d[lag:], d[:-lag], ddof=1)[0, 1]
        long_run_var += 2 * (1 - lag / (max_lag + 1)) * covariance
    if not np.isfinite(long_run_var) or long_run_var <= 0:
        return np.nan, np.nan
    dm_stat = d.mean() / np.sqrt(long_run_var / n)
    harvey_adjustment = np.sqrt((n + 1 - 2 * horizon + horizon * (horizon - 1) / n) / n)
    dm_stat *= harvey_adjustment
    p_value = 2 * stats.t.sf(abs(dm_stat), df=n - 1)
    return dm_stat, p_value

dm_rows = []
lstm_error = residual_by_model["LSTM"]
for comparator in evaluation["Model"]:
    if comparator == "LSTM":
        continue
    stat, p_value = diebold_mariano(lstm_error, residual_by_model[comparator], horizon=1)

## 3. Modelling LSTM

Parameter LSTM ditetapkan sebelum evaluasi test dan identik dengan LSTM pada forecast produksi beras.


--- CELL 32 code ---
POP_SEED = 42
random.seed(POP_SEED); np.random.seed(POP_SEED); torch.manual_seed(POP_SEED)
POP_LSTM_PARAMS = {"lag": 3, "hidden": 8, "lr": 0.01, "weight_decay": 0.0001, "epochs": 250}
print("Parameter LSTM populasi:", POP_LSTM_PARAMS)

def pop_supervised(values, lag):
    values = np.asarray(values, dtype=float)
    return np.array([values[i-lag:i] for i in range(lag, len(values))]), values[lag:]

class PopulationLSTM(nn.Module):
    def __init__(self, hidden):
        super().__init__()
        self.lstm = nn.LSTM(1, hidden, batch_first=True)
        self.out = nn.Linear(hidden, 1)
    def forward(self, x):
        return self.out(self.lstm(x)[0][:, -1, :])

def pop_fit_lstm_and_forecast(history, horizon, params=POP_LSTM_PARAMS):
    scaler = MinMaxScaler()
    scaled = scaler.fit_transform(np.asarray(history, dtype=float).reshape(-1, 1)).ravel()
    X, y = pop_supervised(scaled, params["lag"])
    X_tensor = torch.tensor(X[:, :, None], dtype=torch.float32)
    y_tensor = torch.tensor(y[:, None], dtype=torch.float32)
    torch.manual_seed(POP_SEED)
    model = PopulationLSTM(params["hidden"])
    optimizer = torch.optim.Adam(model.parameters(), lr=params["lr"], weight_decay=params["weight_decay"])
    loss_fn = nn.MSELoss()
    model.train()
    for _ in range(params["epochs"]):
        optimizer.zero_grad(); loss = loss_fn(model(X_tensor), y_tensor); loss.backward(); optimizer.step()
    model.eval(); scaled_history, scaled_forecast = list(scaled), []
    with torch.no_grad():
        for _ in range(horizon):
            features = torch.tensor(np.array(scaled_history[-params["lag"]:])[None, :, None], dtype=torch.float32)
            prediction = model(features).item()
            scaled_history.append(prediction); scaled_forecast.append(prediction)
    return scaler.inverse_transform(np.array(scaled_forecast).reshape(-1, 1)).ravel()

def pop_mape(y_true, y_pred):
    return np.mean(np.abs((np.asarray(y_true) - np.asarray(y_pred)) / np.asarray(y_true))) * 100

# Tuning MLP menggunakan data train populasi saja; 2013–2018 menjadi validasi internal.
from sklearn.neural_network import MLPRegressor
POP_MLP_GRID = [{"lag": lag, "hidden": hidden, "alpha": alpha, "lr": lr}
                for lag in [3, 5] for hidden in [(8,), (16,)]
                for alpha in [.0001, .01] for lr in [.001, .01]]

def pop_mlp_forecast(history, horizon, params):
    scaler = MinMaxScaler()
    scaled = scaler.fit_transform(np.asarray(history, dtype=float).reshape(-1, 1)).ravel()
    X, y = pop_supervised(scaled, params["lag"])
    model = MLPRegressor(hidden_layer_sizes=params["hidden"], alpha=params["alpha"],
                         learning_rate_init=params["lr"], max_iter=3000, early_stopping=True,
                         validation_fraction=.2, random_state=POP_SEED)
    model.fit(X, y)
    scaled_history, scaled_forecast = list(scaled), []
    for _ in range(horizon):
        prediction = model.predict(np.array(scaled_history[-params["lag"]:]).reshape(1, -1))[0]
        scaled_history.append(prediction); scaled_forecast.append(prediction)
    return scaler.inverse_transform(np.array(scaled_forecast).reshape(-1, 1)).ravel()

pop_mlp_tune_y, pop_mlp_val_y = pop_train_y[:-6], pop_train_y[-6:]
